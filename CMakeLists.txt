cmake_minimum_required(VERSION 3.20)
project(rkmoon VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "RKMoon runtime and offline socket tests require Linux")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(RKMOON_MPP "Build the real hardware worker (requires Rockchip MPP SDK)" OFF)
option(RKMOON_SANITIZE "ASan and UBSan for native offline tests" OFF)
find_package(Threads REQUIRED)
add_library(rkmoon_core src/core.cpp src/annexb.cpp src/pixels.cpp src/hid_client.cpp)
target_include_directories(rkmoon_core PUBLIC include)
target_link_libraries(rkmoon_core PUBLIC Threads::Threads)
target_compile_options(rkmoon_core PRIVATE -Wall -Wextra -Wpedantic)
if(RKMOON_SANITIZE)
  target_compile_options(rkmoon_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(rkmoon_core PUBLIC -fsanitize=address,undefined)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_library(rkmoon_capture src/capture.cpp)
  target_link_libraries(rkmoon_capture PUBLIC rkmoon_core)
  target_compile_options(rkmoon_capture PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(rkmoon-v4l2-probe src/probe_main.cpp)
  target_link_libraries(rkmoon-v4l2-probe PRIVATE rkmoon_capture)
endif()
if(RKMOON_MPP)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "MPP worker requires Linux; never substitute a software encoder")
  endif()
  find_path(MPP_INCLUDE_DIR rk_mpi.h PATH_SUFFIXES rockchip mpp REQUIRED)
  find_library(MPP_LIBRARY NAMES rockchip_mpp REQUIRED)
  add_executable(rkmoon-worker src/worker_main.cpp src/encoder_mpp.cpp)
  target_include_directories(rkmoon-worker PRIVATE ${MPP_INCLUDE_DIR})
  target_link_libraries(rkmoon-worker PRIVATE rkmoon_capture ${MPP_LIBRARY})
  target_compile_options(rkmoon-worker PRIVATE -Wall -Wextra -Wpedantic)
  install(TARGETS rkmoon-worker RUNTIME DESTINATION bin)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(rkmoon-hid-fixture tests/hid_client_fixture.cpp)
  target_link_libraries(rkmoon-hid-fixture PRIVATE rkmoon_core)
  add_executable(rkmoon-core-tests tests/core_tests.cpp)
  target_link_libraries(rkmoon-core-tests PRIVATE rkmoon_core)
  foreach(name wire_roundtrip wire_oversize wire_reserved wire_kind wire_payload wire_flags config_valid config_invalid h264_idr hevc_idr hevc_cra missing_headers two_pictures bad_nal sequence_bad sequence_good pixel_black pixel_white pixel_limited_levels pixel_limited_colors pixel_limited_stride pixel_stride pixel_bounds direct_layout ipc_roundtrip ipc_eof ipc_deadline child_spawn)
    add_test(NAME ${name} COMMAND rkmoon-core-tests ${name})
  endforeach()
endif()
